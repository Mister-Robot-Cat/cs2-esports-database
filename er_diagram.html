    countries {
        int     country_id PK
        varchar name UK
        char    iso_code UK
    }
    teams {
        int      team_id PK
        varchar  name UK
        varchar  tag
        int      country_id FK
        smallint founded_year
        int      world_rank UK
    }
    players {
        int      player_id PK
        varchar  nickname UK
        varchar  full_name
        smallint birth_year
        int      country_id FK
        int      team_id FK "NULL = free agent"
        varchar  role
    }
    maps {
        int     map_id PK
        varchar name UK
        varchar code UK
        boolean is_active_duty
    }
    weapons {
        int     weapon_id PK
        varchar name UK
        varchar category
        varchar side
        int     price
        int     damage
        int     magazine_size
    }
    rarities {
        int      rarity_id PK
        varchar  name UK
        char     color_hex
        smallint tier_order UK
    }
    skins {
        int     skin_id PK
        int     weapon_id FK
        int     rarity_id FK
        varchar name
        numeric market_price
    }
    player_skins {
        int     player_id PK, FK
        int     skin_id PK, FK
        numeric float_value
        boolean is_stattrak
        date    acquired_on
    }
    tournaments {
        int     tournament_id PK
        varchar name UK
        varchar organizer
        varchar city
        int     country_id FK
        char    tier
        boolean is_major
        date    start_date
        date    end_date
        numeric prize_pool
    }
    matches {
        int      match_id PK
        int      tournament_id FK
        varchar  stage
        date     match_date
        smallint best_of
        int      team1_id FK
        int      team2_id FK
        int      winner_team_id FK
    }
    match_maps {
        int      match_map_id PK
        int      match_id FK
        int      map_id FK
        smallint map_number
        int      picked_by_team_id FK "NULL = decider"
        smallint team1_score
        smallint team2_score
    }
    player_match_stats {
        int      match_map_id PK, FK
        int      player_id PK, FK
        int      team_id FK
        smallint kills
        smallint deaths
        smallint assists
        smallint headshots
        numeric  adr
    }
