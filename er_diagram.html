    tournaments ||--o{ matches      : "матчи турнира"
    maps        ||--o{ matches      : "карта матча"
    teams       ||--o{ matches      : "team1 / team2 / winner"

    countries {
        int     country_id PK
        varchar name
    }
    teams {
        int     team_id PK
        varchar name
        int     country_id FK
    }
    players {
        int     player_id PK
        varchar nickname
        varchar full_name
        int     age
        varchar role
        int     team_id FK
        int     country_id FK
    }
    tournaments {
        int     tournament_id PK
        varchar name
        varchar city
        int     prize_pool
    }
    maps {
        int     map_id PK
        varchar name
    }
    matches {
        int     match_id PK
        int     tournament_id FK
        int     map_id FK
        int     team1_id FK
        int     team2_id FK
        int     team1_score
        int     team2_score
        int     winner_id FK
        date    match_date
    }
    weapons {
        int     weapon_id PK
        varchar name
        varchar type
        int     price
    }
    skins {
        int     skin_id PK
        int     weapon_id FK
        varchar name
        varchar rarity
        numeric price
    }
    player_skins {
        int     player_id PK, FK
        int     skin_id PK, FK
    }
